cmake_minimum_required(VERSION 3.22)
project(gblink_android C)
set(BUILD_QT OFF CACHE BOOL "" FORCE)
set(BUILD_SDL OFF CACHE BOOL "" FORCE)
set(BUILD_GL OFF CACHE BOOL "" FORCE)
set(BUILD_GLES2 OFF CACHE BOOL "" FORCE)
set(BUILD_GLES3 OFF CACHE BOOL "" FORCE)
set(BUILD_LIBRETRO ON CACHE BOOL "" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(SKIP_LIBRARY ON CACHE BOOL "" FORCE)
set(DISABLE_DEPS ON CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
set(USE_DEBUGGERS OFF CACHE BOOL "" FORCE)
set(M_CORE_GB OFF CACHE BOOL "" FORCE)
set(BUILD_LTO OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_SUITE OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../third_party/mgba mgba)
get_filename_component(MGBA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/mgba" ABSOLUTE)
get_target_property(MGBA_RETRO_SOURCES mgba_libretro SOURCES)
list(REMOVE_ITEM MGBA_RETRO_SOURCES "${MGBA_ROOT}/src/platform/libretro/libretro.c")
file(READ "${MGBA_ROOT}/src/platform/libretro/libretro.c" MGBA_RETRO_TEXT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/companion_libretro.c" "${MGBA_RETRO_TEXT}\n#include \"companion_hooks.h\"\n")
list(APPEND MGBA_RETRO_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/companion_libretro.c")
set_property(TARGET mgba_libretro PROPERTY SOURCES ${MGBA_RETRO_SOURCES})
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MGBA_ROOT}/src/platform/libretro/libretro.c")
target_sources(mgba_libretro PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/bridge.c)
target_include_directories(mgba_libretro PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/mgba/src/platform/libretro)
set_target_properties(mgba_libretro PROPERTIES PREFIX "lib" OUTPUT_NAME "companion")
if(ANDROID)
    target_link_libraries(mgba_libretro log)
    target_link_options(mgba_libretro PRIVATE "-Wl,-z,max-page-size=16384")
else()
    if(WIN32)
        set_target_properties(mgba_libretro PROPERTIES PREFIX "")
    endif()
    # The same JNI frontend can be smoke-tested on a development computer.
    find_package(JNI REQUIRED)
    target_include_directories(mgba_libretro PRIVATE ${JNI_INCLUDE_DIRS})
endif()
